cmake_minimum_required(VERSION 3.13)
project(melonDSOunceBiosProbe LANGUAGES C CXX)
if(NOT CMAKE_CROSSCOMPILING OR NOT DEFINED ENV{DEVKITPRO})
    message(FATAL_ERROR "Source switch2vars.sh and use aarch64-none-elf-cmake")
endif()
add_subdirectory(.. ounce-core)
find_library(OUNCE_RUNTIME NAMES ounce nx PATHS "$ENV{DEVKITPRO}/libounce/lib" NO_DEFAULT_PATH)
find_program(ELF2NRO elf2nro HINTS "$ENV{DEVKITPRO}/tools/bin")
if(NOT OUNCE_RUNTIME OR NOT ELF2NRO)
    message(FATAL_ERROR "Missing libounce runtime or elf2nro")
endif()
add_executable(melonds-ounce-bios-probe main.cpp ../runner/Platform.cpp)
set_target_properties(melonds-ounce-bios-probe PROPERTIES SUFFIX ".elf" CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES)
target_include_directories(melonds-ounce-bios-probe BEFORE PRIVATE "$ENV{DEVKITPRO}/libounce/include" "${CMAKE_CURRENT_LIST_DIR}/../../src")
target_link_libraries(melonds-ounce-bios-probe PRIVATE core ${OUNCE_RUNTIME} m)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/melonds-ounce-bios-probe.nro"
    COMMAND "${ELF2NRO}" $<TARGET_FILE:melonds-ounce-bios-probe> "${CMAKE_CURRENT_BINARY_DIR}/melonds-ounce-bios-probe.nro"
    DEPENDS melonds-ounce-bios-probe VERBATIM)
add_custom_target(ounce-bios-probe-nro ALL DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/melonds-ounce-bios-probe.nro")
